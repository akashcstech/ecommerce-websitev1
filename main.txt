async def seed() -> None:
    await ensure_indexes()
    users = [
        ("Store Admin", "admin@store.com", "9000000001", "admin123", True, "Bazaarline HQ, MG Road, Bengaluru 560001"),
        ("Priya Sharma", "customer@store.com", "9876543210", "customer123", False, "12 Lake View Apartments, Indiranagar, Bengaluru 560038"),
    ]
    for name, email, phone, pw, is_admin, address in users:
        if not await db.users.find_one({"email": email}):
            await db.users.insert_one(
                {
                    "id": new_id(),
                    "name": name,
                    "email": email,
                    "phone": phone,
                    "address": address,
                    "password_hash": hash_password(pw),
                    "is_admin": is_admin,
                    "is_active": True,
                    "created_at": now_iso(),
                }
            )
        else:  # keep demo credentials working
            await db.users.update_one(
                {"email": email},
                {"$set": {"password_hash": hash_password(pw), "is_admin": is_admin, "is_active": True}},
            )

    cat_ids: dict[str, str] = {}
    for name in CATEGORIES:
        doc = await db.categories.find_one({"name": name})
        if not doc:
            doc = {"id": new_id(), "name": name}
            await db.categories.insert_one(dict(doc))
        cat_ids[name] = doc["id"]

    for name, cat, price, stock, img, desc in PRODUCTS:
        if await db.products.find_one({"name": name}):
            continue
        await db.products.insert_one(
            {
                "id": new_id(),
                "category_id": cat_ids[cat],
                "name": name,
                "description": desc,
                "price": float(price),
                "stock": stock,
                "image_url": f"{U}{img}{Q}",
                "created_at": now_iso(),
            }
        )
    print(f"Seeded: {await db.users.count_documents({})} users, {await db.categories.count_documents({})} categories, {await db.products.count_documents({})} products")


if __name__ == "__main__":
    asyncio.run(seed())

"""Password hashing + httpOnly cookie sessions (JWT) + FastAPI auth dependencies."""

import os
from datetime import datetime, timedelta, timezone

import bcrypt
import jwt
from fastapi import Cookie, Depends, HTTPException, Response

from lib.db import db  # also loads backend/.env

JWT_SECRET = os.environ["JWT_SECRET"]
COOKIE_NAME = "session"
SESSION_DAYS = 7


def hash_password(password: str) -> str:
    return bcrypt.hashpw(password.encode(), bcrypt.gensalt()).decode()


def verify_password(password: str, hashed: str) -> bool:
    try:
        return bcrypt.checkpw(password.encode(), hashed.encode())
    except ValueError:
        return False


def set_session(response: Response, user_id: str) -> None:
    exp = datetime.now(timezone.utc) + timedelta(days=SESSION_DAYS)
    token = jwt.encode({"sub": user_id, "exp": exp}, JWT_SECRET, algorithm="HS256")
    response.set_cookie(
        COOKIE_NAME, token, httponly=True, samesite="lax", max_age=SESSION_DAYS * 86400, path="/"
    )


def clear_session(response: Response) -> None:
    response.delete_cookie(COOKIE_NAME, path="/")


async def current_user(session: str | None = Cookie(default=None)) -> dict:
    if not session:
        raise HTTPException(status_code=401, detail="Please log in")
    try:
        payload = jwt.decode(session, JWT_SECRET, algorithms=["HS256"])
    except jwt.PyJWTError:
        raise HTTPException(status_code=401, detail="Session expired, please log in again")
    user = await db.users.find_one({"id": payload.get("sub")}, {"_id": 0})
    if not user:
        raise HTTPException(status_code=401, detail="Account not found")
    if not user.get("is_active", True):
        raise HTTPException(status_code=403, detail="Your account has been deactivated")
    return user


async def current_customer(user: dict = Depends(current_user)) -> dict:
    if user.get("is_admin"):
        raise HTTPException(status_code=403, detail="Admins cannot shop with the admin account")
    return user


async def current_admin(user: dict = Depends(current_user)) -> dict:
    if not user.get("is_admin"):
        raise HTTPException(status_code=403, detail="Admin access only")
    return user

import logging
import os
from pathlib import Path

from dotenv import load_dotenv
from motor.motor_asyncio import AsyncIOMotorClient
from pymongo import ASCENDING, DESCENDING, IndexModel

load_dotenv(Path(__file__).parent.parent / ".env")

# One entry per collection: every field a route filters, sorts, or dedupes on. Applied by ensure_indexes() at startup.
INDEXES: dict[str, list[IndexModel]] = {
    "users": [IndexModel([("id", ASCENDING)], name="id", unique=True), IndexModel([("email", ASCENDING)], name="email", unique=True)],
    "categories": [IndexModel([("id", ASCENDING)], name="id", unique=True), IndexModel([("name", ASCENDING)], name="name")],
    "products": [
        IndexModel([("id", ASCENDING)], name="id", unique=True),
        IndexModel([("name", ASCENDING)], name="name"),
        IndexModel([("category_id", ASCENDING), ("created_at", DESCENDING)], name="category_created"),
        IndexModel([("price", ASCENDING)], name="price"),
    ],
    "carts": [IndexModel([("id", ASCENDING)], name="id", unique=True), IndexModel([("user_id", ASCENDING), ("product_id", ASCENDING)], name="user_product", unique=True)],
    "orders": [IndexModel([("id", ASCENDING)], name="id", unique=True), IndexModel([("user_id", ASCENDING), ("created_at", DESCENDING)], name="user_created"), IndexModel([("status", ASCENDING), ("created_at", DESCENDING)], name="status_created")],
}


async def ensure_indexes() -> None:
    for collection, models in INDEXES.items():
        for model in models:  # one at a time so a bad spec skips only itself
            try:
                await db[collection].create_indexes([model])
            except Exception as exc:  # never block boot on an index; the log line names what to fix
                logger.error("ensure_indexes(%s.%s): %s", collection, model.document["name"], exc)

# ---------- Dashboard ----------
@router.get("/stats", response_model=AdminStats)
async def stats():
    revenue_rows = await db.orders.aggregate(
        [{"$match": {"status": {"$ne": "Cancelled"}}}, {"$group": {"_id": None, "sum": {"$sum": "$total"}}}]
    ).to_list(1)
    return AdminStats(
        orders=await db.orders.count_documents({}),
        products=await db.products.count_documents({}),
        customers=await db.users.count_documents({"is_admin": False}),
        categories=await db.categories.count_documents({}),
        revenue=round(revenue_rows[0]["sum"], 2) if revenue_rows else 0,
        placed_orders=await db.orders.count_documents({"status": "Placed"}),
        out_of_stock=await db.products.count_documents({"stock": {"$lte": 0}}),
    )


# ---------- Categories ----------
async def _unique_category(name: str, exclude_id: str | None = None) -> None:
    import re

    q: dict = {"name": {"$regex": f"^{re.escape(name)}$", "$options": "i"}}
    if exclude_id:
        q["id"] = {"$ne": exclude_id}
    if await db.categories.find_one(q):
        raise HTTPException(status_code=400, detail="A category with this name already exists")


@router.post("/categories", response_model=Category)
async def create_category(body: CategoryIn):
    name = body.name.strip()
    await _unique_category(name)
    cat = {"id": new_id(), "name": name}
    await db.categories.insert_one(dict(cat))
    return Category(**cat)


@router.put("/categories/{id}", response_model=Category)
async def update_category(id: str, body: CategoryIn):
    name = body.name.strip()
    await _unique_category(name, exclude_id=id)
    res = await db.categories.update_one({"id": id}, {"$set": {"name": name}})
    if res.matched_count != 1:
        raise HTTPException(status_code=404, detail="Category not found")
    return Category(id=id, name=name, product_count=await db.products.count_documents({"category_id": id}))


@router.delete("/categories/{id}")

async def build_cart(user_id: str) -> Cart:
    rows = await db.carts.find({"user_id": user_id}, {"_id": 0}).sort("added_at", 1).to_list(500)
    ids = [r["product_id"] for r in rows]
    products = {p["id"]: p async for p in db.products.find({"id": {"$in": ids}}, {"_id": 0})}
    cats = await category_map()
    items: list[CartItem] = []
    for r in rows:
        p = products.get(r["product_id"])
        if not p:  # product was deleted by admin — drop the stale row
            await db.carts.delete_one({"id": r["id"]})
            continue
        product = to_product(p, cats)
        items.append(
            CartItem(
                id=r["id"],
                product_id=r["product_id"],
                quantity=r["quantity"],
                product=product,
                line_total=round(product.price * r["quantity"], 2),
            )
        )
    return Cart(
        items=items,
        total=round(sum(i.line_total for i in items), 2),
        count=sum(i.quantity for i in items),
    )

        raise HTTPException(status_code=400, detail="This product is out of stock")
    existing = await db.carts.find_one({"user_id": user["id"], "product_id": body.product_id})
    new_qty = body.quantity + (existing["quantity"] if existing else 0)
    if new_qty > product["stock"]:

    if existing:
        await db.carts.update_one({"id": existing["id"]}, {"$set": {"quantity": new_qty}})
    else:
        await db.carts.insert_one(
            {
                "id": new_id(),
                "user_id": user["id"],
                "product_id": body.product_id,
                "quantity": new_qty,

            }

@router.get("/products", response_model=ProductPage)
async def list_products(
    q: str = "",
    category_id: str = "",
    sort: Literal["newest", "price_asc", "price_desc", "name"] = "newest",
    in_stock: bool = False,
    page: int = Query(1, ge=1),

        query["name"] = {"$regex": re.escape(q.strip()), "$options": "i"}

"""GET /api/main-txt — builds main.txt live from the real project source files.

Format (strict):
  ROOT FOLDER STRUCTURE (root/ tree)
  then for every file:
    - FILE NAME: <path>
    - COPYPASTE
    <file content>
    - CODE -END
"""

from datetime import datetime, timezone
from pathlib import Path

from fastapi import APIRouter
from fastapi.responses import PlainTextResponse

router = APIRouter(tags=["export"])

ROOT = Path(__file__).resolve().parents[2]  # /app
MAIN_TXT = ROOT / "main.txt"

# Source written for this app (preinstalled shadcn UI kit, lockfiles and tool config excluded).
INCLUDE_GLOBS = [
    "backend/server.py",
    "backend/seed.py",
    "backend/requirements.txt",
    "backend/.env.example",
    "backend/lib/*.py",
    "backend/models/*.py",
    "backend/routers/*.py",
    "frontend/index.html",
    "frontend/src/**/*.ts",
    "frontend/src/**/*.tsx",
    "frontend/src/**/*.css",
]
EXCLUDE_PARTS = {"__pycache__", "node_modules", "ui"}
EXCLUDE_FILES = {"frontend/src/lib/recharts.tsx", "frontend/src/lib/lucide-react.tsx", "frontend/src/vite-env.d.ts"}


def collect_files() -> list[str]:
    files: set[str] = set()
    for pattern in INCLUDE_GLOBS:
        for p in ROOT.glob(pattern):
            rel = p.relative_to(ROOT).as_posix()
            if p.is_file() and not (EXCLUDE_PARTS & set(p.relative_to(ROOT).parts)) and rel not in EXCLUDE_FILES:
                files.add(rel)
    # backend first, then frontend; folders before their nested folders' files read naturally
    return sorted(files, key=lambda f: (f.split("/")[0] != "backend", f.count("/"), f))


def build_tree(files: list[str]) -> str:
    tree: dict = {}
    for f in files:
        node = tree
        for part in f.split("/"):
            node = node.setdefault(part, {})
    lines = ["root/"]

    def walk(node: dict, prefix: str) -> None:
        entries = sorted(node.items(), key=lambda kv: (not kv[1], kv[0]))  # folders first
        for i, (name, child) in enumerate(entries):
            last = i == len(entries) - 1
            lines.append(f"{prefix}{'└── ' if last else '├── '}{name}{'/' if child else ''}")
            if child:
                walk(child, prefix + ("    " if last else "│   "))

    walk(tree, "")
    return "\n".join(lines)


def build_main_txt() -> str:
    files = collect_files()
    out = [
        "MAIN.TXT - Online Shopping Website (Bazaarline) - full source export",
        f"Generated: {datetime.now(timezone.utc).strftime('%Y-%m-%d %H:%M:%S UTC')}",
        f"Total files: {len(files)}",
        "",
        "1) ROOT FOLDER STRUCTURE",
        "",
        build_tree(files),
        "",
        "2) FILES",
        "",
    ]
    for rel in files:
        content = (ROOT / rel).read_text(encoding="utf-8", errors="replace").rstrip("\n")
        out += [f"- FILE NAME: root/{rel}", "- COPYPASTE", content, "- CODE -END", ""]
    out.append("- END OF main.txt")
    return "\n".join(out) + "\n"


@router.get("/main-txt", response_class=PlainTextResponse)
async def download_main_txt():
    text = build_main_txt()
    MAIN_TXT.write_text(text, encoding="utf-8")  # keep /app/main.txt on disk in sync too
    return PlainTextResponse(
        text,
        headers={"Content-Disposition": 'attachment; filename="main.txt"', "Cache-Control": "no-store"},
    )

def validate_card(card: CardIn | None) -> str:
    """Simulated payment: validate the card shape and return its last 4 digits. No card data is stored."""
    if card is None:
        raise HTTPException(status_code=400, detail="Card details are required for card payment")
    digits = re.sub(r"[\s-]", "", card.number)
    if not digits.isdigit() or not 12 <= len(digits) <= 19:
        raise HTTPException(status_code=400, detail="Card number must be 12-19 digits")
    month, year = card.expiry.split("/")
    now = datetime.now(timezone.utc)
    if (2000 + int(year), int(month)) < (now.year, now.month):
        raise HTTPException(status_code=400, detail="Card has expired")
    if digits.endswith("0000"):
        raise HTTPException(status_code=402, detail="Payment declined by the bank (simulated)")
    return digits[-4:]


async def restore_stock(items: list[dict]) -> None:
    for it in items:
        await db.products.update_one({"id": it["product_id"]}, {"$inc": {"stock": it["quantity"]}})


@router.post("/checkout", response_model=Order)
async def checkout(body: CheckoutIn, user: dict = Depends(current_customer)):
    cart = await build_cart(user["id"])
    if not cart.items:
        raise HTTPException(status_code=400, detail="Your cart is empty")
    card_last4 = validate_card(body.card) if body.payment_method == "card" else None

    # Reserve stock item by item with a conditional decrement; roll back on any failure
    # so a failed order leaves no partial data (two buyers can't both take the last unit).
    reserved: list[dict] = []
    for item in cart.items:
        res = await db.products.update_one(
            {"id": item.product_id, "stock": {"$gte": item.quantity}},
            {"$inc": {"stock": -item.quantity}},
        )
        if res.modified_count != 1:
            await restore_stock(reserved)
            raise HTTPException(
                status_code=400,
                detail=f"'{item.product.name}' does not have enough stock. Please update your cart.",
            )
        reserved.append({"product_id": item.product_id, "quantity": item.quantity})

    order = {
        "id": new_id(),
        "user_id": user["id"],
        "user_name": user["name"],
        "user_email": user["email"],
        "items": [
            {
                "id": new_id(),
                "product_id": i.product_id,
                "name": i.product.name,
                "image_url": i.product.image_url,
                "quantity": i.quantity,
                "unit_price": i.product.price,
            }
            for i in cart.items
        ],
        "total": cart.total,
        "status": "Placed",
        "payment_method": body.payment_method,
        "payment_status": "paid" if body.payment_method == "card" else "pending",
        "card_last4": card_last4,
        "delivery_address": body.delivery_address.strip(),
        "created_at": now_iso(),
    }
    try:
        await db.orders.insert_one(dict(order))
    except Exception:
        await restore_stock(reserved)
        raise HTTPException(status_code=500, detail="Could not place order, please retry")
    await db.carts.delete_many({"user_id": user["id"]})
    if not user.get("address"):
        await db.users.update_one({"id": user["id"]}, {"$set": {"address": order["delivery_address"]}})
    return Order(**order)

import { useMutation, useQueryClient } from "@tanstack/react-query";
import { toast } from "sonner";
import { Sheet, SheetContent, SheetDescription, SheetHeader, SheetTitle } from "@/components/ui/sheet";

export default function CartDrawer({ open, onOpenChange }: { open: boolean; onOpenChange: (o: boolean) => void }) {
  const qc = useQueryClient();
  const navigate = useNavigate();
  const { data: cart } = useCart();

  const update = useMutation({
    mutationFn: (v: { id: string; quantity: number }) => apiPatch<Cart>(`/cart/items/${v.id}`, { quantity: v.quantity }),
    onSuccess: (c) => qc.setQueryData(["cart"], c),
    onError: (e) => toast.error(errMsg(e)),
  });
  const remove = useMutation({
    mutationFn: (id: string) => apiDelete<Cart>(`/cart/items/${id}`),
    onSuccess: (c) => {
      qc.setQueryData(["cart"], c);
      toast.success("Removed from cart");
    },
    onError: (e) => toast.error(errMsg(e)),
  });

  const items = cart?.items ?? [];

  return (
    <Sheet open={open} onOpenChange={onOpenChange}>
      <SheetContent side="right" className="flex w-full flex-col gap-0 p-0 sm:max-w-md" data-testid="cart-drawer">
        <SheetHeader className="border-b px-6 py-5">
          <SheetTitle className="font-heading text-xl">Your cart</SheetTitle>
          <SheetDescription data-testid="cart-item-count">
            {cart?.count ?? 0} item{(cart?.count ?? 0) === 1 ? "" : "s"}
          </SheetDescription>
        </SheetHeader>

        <div className="flex-1 overflow-y-auto px-6 py-4">
          {items.length === 0 ? (
            <div className="flex h-full flex-col items-center justify-center gap-3 text-center" data-testid="cart-empty-state">
              <ShoppingBag className="size-10 text-slate-300" />
              <p className="font-medium">Your cart is empty</p>
              <p className="text-sm text-muted-foreground">Browse the catalogue and add something you love.</p>
            </div>
          ) : (
            <ul className="divide-y">
              {items.map((it) => (
                <li key={it.id} className="flex gap-4 py-4" data-testid={`cart-item-${it.product_id}`}>
                  <img src={it.product.image_url} alt={it.product.name} className="size-20 rounded-xl object-cover" />
                  <div className="flex min-w-0 flex-1 flex-col gap-2">
                    <div className="flex items-start justify-between gap-2">
                      <Link
                        to={`/product/${it.product_id}`}
                        onClick={() => onOpenChange(false)}
                        className="line-clamp-2 text-sm font-medium hover:underline"
                      >
                        {it.product.name}
                      </Link>
                      <button
                        type="button"
                        aria-label="Remove item"
                        data-testid={`cart-remove-${it.product_id}`}
                        onClick={() => remove.mutate(it.id)}
                        className="rounded-md p-1 text-slate-400 transition-colors hover:bg-red-50 hover:text-red-600"
                      >
                        <Trash2 className="size-4" />
                      </button>

                      <div className="flex items-center rounded-full border">
                        <button
                          type="button"
                          aria-label="Decrease quantity"
                          data-testid={`cart-decrease-${it.product_id}`}
                          disabled={it.quantity <= 1 || update.isPending}

                          className="p-2 transition-colors hover:bg-slate-100 disabled:opacity-40"
                        >
                          <Minus className="size-3" />

// Typed fetch layer over the FastAPI backend. Base is the relative "/api" prefix so the
// same code works in dev (Vite proxies /api → :8001) and behind a single origin in prod.
const BASE = "/api";

// Fields are declared, not constructor parameter properties: tsconfig sets
// erasableSyntaxOnly, which rejects `constructor(readonly status: number)`.
export class ApiError extends Error {
  status: number;
  body: unknown;

  constructor(status: number, body: unknown) {
    super(`request failed with ${status}`);
    this.name = "ApiError";
    this.status = status;
    this.body = body;
  }
}

type JsonBody = unknown;

async function request<T>(method: string, path: string, body?: JsonBody): Promise<T> {
  // Auth rides the httpOnly session cookie automatically — never add auth headers here.
  const res = await fetch(`${BASE}${path}`, {
    method,
    headers: body === undefined ? undefined : { "Content-Type": "application/json" },
    body: body === undefined ? undefined : JSON.stringify(body),
  });

  // FastAPI reports request-validation failures as 422 with a {detail: [...]} body.
  if (!res.ok) {
    const errBody = await res.json().catch(() => null);
    throw new ApiError(res.status, errBody);
  }

  if (res.status === 204) return undefined as T;
  return (await res.json()) as T;
}

// The response type is yours to declare: nothing infers across the Python boundary, so a
// TS interface here mirrors the endpoint's Pydantic model by hand — keep the two in sync.
export const apiGet = <T>(path: string) => request<T>("GET", path);
export const apiPost = <T>(path: string, body?: JsonBody) => request<T>("POST", path, body ?? null);
export const apiPut = <T>(path: string, body?: JsonBody) => request<T>("PUT", path, body ?? null);
export const apiPatch = <T>(path: string, body?: JsonBody) =>
  request<T>("PATCH", path, body ?? null);
export const apiDelete = <T>(path: string) => request<T>("DELETE", path);

  currency: "INR",

// Turns FastAPI error bodies ({detail: string} or 422 {detail: [...]}) into a readable message.
export function errMsg(e: unknown, fallback = "Something went wrong. Please try again."): string {
  if (e instanceof ApiError) {
    const body = e.body as { detail?: unknown } | null;
    if (typeof body?.detail === "string") return body.detail;
    if (Array.isArray(body?.detail) && body.detail.length > 0) {
      const first = body.detail[0] as { msg?: string; loc?: unknown[] };
      const field = first.loc ? String(first.loc[first.loc.length - 1]).replace(/_/g, " ") : "";
      return field ? `${field}: ${first.msg ?? "invalid"}` : (first.msg ?? fallback);
    }
  }
  return fallback;
}

// Session boundary: auth is an httpOnly cookie the backend owns; the frontend's one
// duty is wiping the react-query cache so one account's data never renders for the next.
import { queryClient } from "./queryClient";

// Call after every successful login/signup.

  queryClient.clear();
}

// Call from every sign-out control; the hard redirect resets all in-memory state.
export async function endSession(redirectTo: string = "/login"): Promise<void> {
  try {
    await apiPost("/auth/logout");
  } finally {
    queryClient.clear();
    window.location.assign(redirectTo);
  }

import { useState, type ChangeEvent, type FormEvent } from "react";
import { keepPreviousData, useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { Loader2, Pencil, Plus, Search, Trash2, Upload } from "lucide-react";
import { toast } from "sonner";
import { Button } from "@/components/ui/button";
import { Dialog, DialogContent, DialogFooter, DialogHeader, DialogTitle } from "@/components/ui/dialog";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table";
import { Textarea } from "@/components/ui/textarea";
import { StockBadge } from "@/components/Badges";
import { useCategories } from "@/hooks/useShop";
import { apiDelete, apiGet, apiPost, apiPut } from "@/lib/api";
import { errMsg, inr } from "@/lib/format";
import type { Product, ProductInput, ProductPage } from "@/lib/types";

const EMPTY = { category_id: "", name: "", description: "", price: "", stock: "", image_url: "" };
type FormState = typeof EMPTY;

export default function AdminProducts() {
  const qc = useQueryClient();
  const [q, setQ] = useState("");
  const [editing, setEditing] = useState<Product | null>(null);
  const [open, setOpen] = useState(false);
  const [form, setForm] = useState<FormState>(EMPTY);
  const { data: categories } = useCategories();

  const list = useQuery({
    queryKey: ["products", { admin: true, q }],
    queryFn: () => apiGet<ProductPage>(`/products?${new URLSearchParams({ q, sort: "name", page_size: "200" })}`),
    placeholderData: keepPreviousData,
  });

  const refresh = () => {
    qc.invalidateQueries({ queryKey: ["products"] });
    qc.invalidateQueries({ queryKey: ["product"] });
    qc.invalidateQueries({ queryKey: ["categories"] });
    qc.invalidateQueries({ queryKey: ["admin"] });
  };

  const save = useMutation({
    mutationFn: (body: ProductInput) =>
      editing ? apiPut<Product>(`/admin/products/${editing.id}`, body) : apiPost<Product>("/admin/products", body),
    onSuccess: (p) => {
      toast.success(editing ? `Updated “${p.name}”` : `Added “${p.name}”`);
      setOpen(false);
      refresh();
    },
    onError: (e) => toast.error(errMsg(e)),
  });
  const remove = useMutation({
    mutationFn: (id: string) => apiDelete<{ ok: boolean }>(`/admin/products/${id}`),
    onSuccess: () => {
      toast.success("Product deleted");
      refresh();
    },
    onError: (e) => toast.error(errMsg(e)),
  });

  const openNew = () => {
    setEditing(null);
    setForm({ ...EMPTY, category_id: categories?.[0]?.id ?? "" });
    setOpen(true);
  };
  const openEdit = (p: Product) => {
    setEditing(p);
    setForm({
      category_id: p.category_id,
      name: p.name,
      description: p.description,
      price: String(p.price),
      stock: String(p.stock),
      image_url: p.image_url,
    });
    setOpen(true);
  };

  const onFile = (e: ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    if (file.size > 1024 * 1024) {
      toast.error("Image must be under 1 MB");
      return;
    }
    const reader = new FileReader();
    reader.onload = () => setForm((f) => ({ ...f, image_url: String(reader.result) }));
    reader.readAsDataURL(file);
  };

  const submit = (e: FormEvent) => {
    e.preventDefault();
    if (!form.category_id) return toast.error("Choose a category");
    save.mutate({
      category_id: form.category_id,
      name: form.name.trim(),
      description: form.description.trim(),
      price: Number(form.price),
      stock: Math.floor(Number(form.stock)),
      image_url: form.image_url.trim(),
    });
  };

  const catName = (id: string) => categories?.find((c) => c.id === id)?.name ?? "Select category";
  const items = list.isError ? [] : (list.data?.items ?? []);

  return (
    <div className="space-y-6">
      <div className="flex flex-wrap items-end justify-between gap-4">
        <div>
          <h1 className="text-3xl font-semibold">Products</h1>
          <p className="text-slate-500">{list.data?.total ?? 0} products in the catalogue</p>
        </div>
        <div className="flex items-center gap-2">
          <div className="relative">
            <Search className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-slate-400" />
            <Input value={q} onChange={(e) => setQ(e.target.value)} placeholder="Search products" className="w-56 rounded-full pl-9" data-testid="admin-product-search-input" />
          </div>
          <Button className="rounded-full" onClick={openNew} data-testid="admin-add-product-button">
            <Plus className="size-4" /> Add product
          </Button>
        </div>
      </div>

      <div className="rounded-2xl border border-slate-200 bg-white">
        <Table>
          <TableHeader>
            <TableRow>
              <TableHead>Product</TableHead>
              <TableHead>Category</TableHead>
              <TableHead className="text-right">Price</TableHead>
              <TableHead className="text-right">Stock</TableHead>
              <TableHead className="text-right">Actions</TableHead>
            </TableRow>
          </TableHeader>
          <TableBody>
            {items.map((p) => (
              <TableRow key={p.id} data-testid={`admin-product-row-${p.id}`}>
                <TableCell>
                  <div className="flex items-center gap-3">
                    <img src={p.image_url} alt="" className="size-10 rounded-lg object-cover" />
                    <span className="max-w-64 truncate font-medium">{p.name}</span>
                  </div>
                </TableCell>
                <TableCell className="text-slate-600">{p.category_name}</TableCell>
                <TableCell className="text-right tabular-nums">{inr(p.price)}</TableCell>
                <TableCell className="text-right">
                  <div className="flex items-center justify-end gap-2">
                    <span className="tabular-nums" data-testid={`admin-product-stock-${p.id}`}>{p.stock}</span>
                    <StockBadge stock={p.stock} />
                  </div>
                </TableCell>
                <TableCell className="text-right">
                  <div className="flex justify-end gap-1">
                    <Button variant="ghost" size="icon-sm" onClick={() => openEdit(p)} aria-label="Edit" data-testid={`admin-edit-product-${p.id}`}>
                      <Pencil className="size-4" />
                    </Button>
                    <Button
                      variant="ghost"
                      size="icon-sm"
                      className="text-red-600 hover:bg-red-50"
                      aria-label="Delete"
                      data-testid={`admin-delete-product-${p.id}`}
                      onClick={() => {
                        if (window.confirm(`Delete “${p.name}”?`)) remove.mutate(p.id);
                      }}
                    >
                      <Trash2 className="size-4" />
                    </Button>
                  </div>
                </TableCell>
              </TableRow>
            ))}
            {items.length === 0 && (
              <TableRow>
                <TableCell colSpan={5} className="py-10 text-center text-slate-500">No products found.</TableCell>
              </TableRow>
            )}
          </TableBody>
        </Table>
      </div>

      <Dialog open={open} onOpenChange={setOpen}>
        <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-lg" data-testid="product-form-dialog">
          <DialogHeader>
            <DialogTitle>{editing ? "Edit product" : "Add product"}</DialogTitle>
          </DialogHeader>
          <form onSubmit={submit} className="space-y-4">
            <div className="space-y-2">
              <Label htmlFor="pname">Name</Label>
              <Input id="pname" required value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} data-testid="product-form-name-input" />
            </div>
            <div className="space-y-2">
              <Label>Category</Label>
              <Select value={form.category_id} onValueChange={(v: string) => setForm({ ...form, category_id: v })}>
                <SelectTrigger className="w-full" data-testid="product-form-category-select">
                  <SelectValue>{(v) => catName(v as string)}</SelectValue>
                </SelectTrigger>
                <SelectContent>
                  {(categories ?? []).map((c) => (
                    <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
                  ))}

              </Select>
            </div>
            <div className="grid grid-cols-2 gap-4">
              <div className="space-y-2">
